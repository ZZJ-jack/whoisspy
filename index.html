'#f7fafc' : ''; dom.roomIdInput.style.background = '#f7fafc'; // 恢复按钮状态 enableGameOper(true, gameConfig.isOwner && !gameConfig.isLock); // 已获取词语则禁用查询按钮 if (gameConfig.hasGetWord) { dom.queryBtn.classList.add('btn-disabled'); dom.queryBtn.disabled = true; } } } // 初始化操作 checkBackendStatus(); // 改为检查后端状态 bindAllEvents(); // 限制房间号仅输入数字 dom.roomIdInput.oninput = function() { this.value = this.value.replace(/\D/g, ''); }; }; // 1. 绑定所有事件 function bindAllEvents() { dom.createRoomBtn.addEventListener('click', createRoom); dom.createRoomBtn.disabled = true; dom.joinRoomBtn.disabled = true; dom.resetBtn.disabled = true; dom.randomBtn.disabled = true; dom.queryBtn.disabled = true; // 加灰样式 dom.createRoomBtn.classList.add('btn-disabled'); dom.joinRoomBtn.classList.add('btn-disabled'); dom.resetBtn.classList.add('btn-disabled'); dom.randomBtn.classList.add('btn-disabled'); dom.queryBtn.classList.add('btn-disabled'); } // 3. 封装请求结束：恢复按钮状态 function endRequest() { isRequesting = false; dom.createRoomBtn.disabled = false; dom.joinRoomBtn.disabled = false; dom.resetBtn.disabled = false; dom.createRoomBtn.classList.remove('btn-disabled'); dom.joinRoomBtn.classList.remove('btn-disabled'); dom.resetBtn.classList.remove('btn-disabled'); // 恢复游戏操作按钮 if (gameConfig.isInRoom) { dom.numInput.disabled = false; if (gameConfig.isOwner && !gameConfig.isLock) { dom.randomBtn.disabled = false; dom.randomBtn.classList.remove('btn-disabled'); } else { dom.randomBtn.disabled = true; dom.randomBtn.classList.add('btn-disabled'); } if (!gameConfig.hasGetWord) { dom.queryBtn.disabled = false; dom.queryBtn.classList.remove('btn-disabled'); } else { dom.queryBtn.disabled = true; dom.queryBtn.classList.add('btn-disabled'); } } else { dom.numInput.disabled = true; dom.randomBtn.disabled = true; dom.queryBtn.disabled = true; dom.randomBtn.classList.add('btn-disabled'); dom.queryBtn.classList.add('btn-disabled'); } } // 4. 封装游戏状态保存 function saveGameConfig() { localStorage.setItem('spyGameConfig', JSON.stringify(gameConfig)); } // 5. 检查后端状态 async function checkBackendStatus() { try { const res = await axios.post(`${BASE_URL}/api/getBankInfo`); if (res.data.code === 0) { const { total, minNum, maxNum } = res.data.data; dom.bankCount.innerText = `(${total}题)`; dom.numInput.min = minNum; dom.numInput.max = maxNum; dom.numInput.placeholder = `输入${minNum}-${maxNum}的编号`; renderResult('success', `✅ 后端题库就绪，共${total}题`); } } catch (err) { console.log('后端题库信息获取失败，使用默认范围'); // 使用默认范围 dom.numInput.placeholder = '输入题目编号'; renderResult('success', '✅ 系统已就绪，可创建/加入房间'); } } // 6. 房主创建房间 - 对接后端 async function createRoom() { if (isRequesting) return; const total = parseInt(dom.playerCount.value.trim()); if (isNaN(total) || total < 2 || total > 20) { renderResult('error', '❌ 总玩家数必须2~20之间，请重新输入'); dom.playerCount.focus(); return; } startRequest(); try { const res = await axios.post(`${BASE_URL}/api/createRoom`, { total }); if (res.data.code !== 0) { return renderResult('error', res.data.msg); } const { roomId } = res.data.data; // 更新全局配置 gameConfig = { isInRoom: true, roomId, isOwner: true, total, lockNum: 0, isLock: false, hasGetWord: false }; saveGameConfig(); // UI反馈 dom.roomIdInput.value = roomId; dom.playerCount.readOnly = true; dom.playerCount.style.background = '#f7fafc'; enableGameOper(true, true); // 房主启用所有出题按钮 renderResult('success', `✅ 房间创建成功！\n你的6位房间号：\n${roomId} \n请告知玩家输入此号加入，\n请点击【房主随机选题】开始游戏` ); const roomId = dom.roomIdInput.value.trim(); if (!roomId || roomId.length !== 6) { renderResult('error', '❌ 请输入6位数字房间号'); dom.roomIdInput.focus(); return; } startRequest(); try { const res = await axios.post(`${BASE_URL}/api/joinRoom`, { roomId }); if (res.data.code !== 0) { return renderResult('error', res.data.msg); } const { isLock, lockNum, total } = res.data.data; // 更新全局配置 gameConfig = { isInRoom: true, roomId, isOwner: false, total, lockNum, isLock, hasGetWord: false }; saveGameConfig(); // 玩家UI逻辑 dom.roomIdInput.readOnly = true; dom.roomIdInput.style.background = '#f7fafc'; if (isLock) { dom.numInput.value = lockNum; dom.numInput.readOnly = true; enableGameOper(true, false); // 玩家禁用随机，启用查询 renderResult('success', `✅ 加入房间成功！\n房间总人数：${total}人\n房主已锁定题目编号：${lockNum}\n点击【获取专属词语】领取你的词` ); } else { enableGameOper(false, false); renderResult('success', `✅ 加入房间成功！\n房间总人数：${total}人\n请等待房主选题并锁定后再操作` ); } dom.lockTip.innerText = isLock ? `（已锁定编号：${lockNum}）` : '（房主尚未选题）'; } catch (err) { renderResult('error', '❌ 加入房间失败，请检查后端或房间号'); console.error(err); } finally { endRequest(); } } // 8. 房主随机选题并锁定 - 对接后端 async function randomAndLockNum() { if (isRequesting) return; if (!gameConfig.isInRoom || !gameConfig.isOwner) { renderResult('error', '❌ 只有房主可以选题！'); return; } // 获取随机题目编号（向后端请求） startRequest(); try { // 先获取题库范围 const rangeRes = await axios.post(`${BASE_URL}/api/getBankInfo`); if (!isNaN(inputNum) && inputNum >= minNum && inputNum <= maxNum) { lockNum = inputNum; } else { // 否则生成随机编号 lockNum = Math.floor(Math.random() * (maxNum - minNum + 1)) + minNum; dom.numInput.value = lockNum; } // 锁定题目 const lockRes = await axios.post(`${BASE_URL}/api/lockNum`, { roomId: gameConfig.roomId, num: lockNum }); if (lockRes.data.code !== 0) { return renderResult('error', lockRes.data.msg); } // 更新配置 gameConfig.lockNum = lockNum; gameConfig.isLock = true; saveGameConfig(); // UI更新 dom.numInput.value = lockNum; dom.numInput.readOnly = true; enableGameOper(true, false); // 锁定后禁用随机选题 renderResult('success', `✅ 选题成功！\n已锁定题目编号：${lockNum}\n所有玩家将强制使用此题，请依次点击【获取专属词语】` ); dom.lockTip.innerText = `（已锁定题目编号：${lockNum}，所有人强制用此题）`; } catch (err) { renderResult('error', '❌ 编号锁定失败，请检查后端'); console.error(err); } finally { endRequest(); } } // 9. 获取专属词语 - 对接后端分配身份 async function getWord() { if (isRequesting) return; if (!gameConfig.isInRoom) { return renderResult('error', '❌ 请先创建/加入房间'); } if (gameConfig.hasGetWord) { return renderResult('error', '❌ 你已获取过词语，不可重复领取'); } startRequest(); try { const res = await axios.post(`${BASE_URL}/api/getWord`, { roomId: gameConfig.roomId }); // 处理各种错误码 if (res.data.code === -1) { return renderResult('error', res.data.msg); } if (res.data.code === -2) { return renderResult('error', res.data.msg); } if (res.data.code === -3) { renderResult('error', res.data.msg); gameConfig.hasGetWord = true; saveGameConfig(); dom.queryBtn.classList.add('btn-disabled'); dom.queryBtn.disabled = true; return; } if (res.data.code === -4) { return renderResult('error', res.data.msg); } const { currRole, lockNum, word, assigned, total } = res.data.data; const roleName = currRole === 'spy' ? '卧底' : '平民'; // 更新状态 gameConfig.hasGetWord = true; saveGameConfig(); // 展示结果 renderResult('game', lockNum, word, roleName, currRole, assigned, total); // 禁用获取按钮 dom.queryBtn.classList.add('btn-disabled'); dom.queryBtn.disabled = true; } catch (err) { renderResult('error', '❌ 身份获取失败，请稍后再试'); console.error(err); } finally { endRequest(); } } // 10. 重置房间 - 对接后端 async function resetRoom() { if (isRequesting) return; if (!gameConfig.isInRoom) { renderResult('error', '❌ 请先创建/加入房间'); return; } startRequest(); try { // 房主才调用后端重置接口 if (gameConfig.isOwner) { const res = await axios.post(`${BASE_URL}/api/resetRoom`, { roomId: gameConfig.roomId }); if (res.data.code !== 0) { return renderResult('error', res.data.msg); } } // 重置前端状态 localStorage.removeItem('spyGameConfig'); gameConfig = { isInRoom: false, roomId: '', isOwner: false, total: 0, lockNum: 0, isLock: false, hasGetWord: false }; // 重置输入框 dom.playerCount.value = ''; dom.roomIdInput.value = ''; dom.numInput.value = ''; dom.numInput.readOnly = false; dom.numInput.disabled = false; dom.playerCount.readOnly = false; dom.roomIdInput.readOnly = false; console.error(err); } finally { endRequest(); } } // 11. 启用/禁用出题按钮 function enableGameOper(isEnable, canRandom) { dom.numInput.disabled = !isEnable; dom.queryBtn.disabled = !isEnable || gameConfig.hasGetWord; if (isEnable && canRandom) { // 获取题库范围 checkBankRange(); } } // 12. 获取题库范围 async function checkBankRange() { try { const res = await axios.post(`${BASE_URL}/api/getBankInfo`); if (res.data.code === 0) { const { minNum, maxNum } = res.data.data;
